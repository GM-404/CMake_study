std::vector<int> TreeNode::rightSideView(TreeNode*root)
{
    std::vector<int> result;
    //根目录肯定是有的
    TreeNode *curr = root;
    if(root!=nullptr)
    {
    result.push_back(root->val);
    }
    while(curr_left!=nullptr||curr_right!=nullptr)
    {
        TreeNode *curr_left = curr->left;
        TreeNode *curr_right = curr->right;
        if(curr_right!=nullptr)
        {
            result.push_back(curr_right->val);
        }
        else
        {
            result.push_back(curr_left->val);
        }

    }
}

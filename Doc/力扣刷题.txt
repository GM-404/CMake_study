bool TreeNode::isValidBST(TreeNode *root)
{
    //对于左子树，子树的左子节点必须小于左节点，子树的右节点必须在左节点和根节点之间
    //对于右子树，子树的左节点必须在右节点和根节点之间，子树的右子点必须在大于右节点
    if(root == nullptr)
    {
        return true;
    }
    int threshold = root->val; //记录下节点的值传入下一个作为区间判断
    //先判断一下根节点的左右是否符合条件
    if(root->left!= nullptr)
    {
        if(root->left > root->val)
        return false;
    }
    if(root->right!= nullptr)
    {
        if(root->right < root->val)
        return false;
    }
    //对于左子树
    bool n1 =  leftIsValidBST(root->left,  threshold)
    //对于右子树
    bool n2 =  rightIsValidBST(root->right,  threshold)
    if(n1 == true && n2 == true)
    {
        return true;
    }
    else{
         return false;
    }
}
